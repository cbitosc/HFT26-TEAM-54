```python
import re

RANK = {"DEBUG": 0, "INFO": 1, "WARN": 2, "WARNING": 2, "ERROR": 3, "CRITICAL": 4}
LINE = re.compile(
    r"[0-9]{4}-(?:0[1-9]|1[0-2])-(?:0[1-9]|[12][0-9]|3[01])[ \t]+"
    r"(?:[01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9][ \t]+"
    r"\[[ \t]*([A-Za-z]+)[ \t]*\][ \t]+(.*)",
    re.DOTALL,
)


def filter_by_level(text: str, min_level: str) -> list:
    if not isinstance(text, str) or not isinstance(min_level, str):
        return []
    threshold = RANK.get(min_level.strip().upper())
    if threshold is None:
        return []
    result = []
    for line in text.splitlines():
        match = LINE.fullmatch(line.strip())
        if match and RANK.get(match.group(1).upper(), -1) >= threshold:
            result.append(match.group(2).strip())
    return result
```
